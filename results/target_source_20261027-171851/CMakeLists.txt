cmake_minimum_required(VERSION 3.18)
project(h3d_route LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(H3D_CUDA "Build the optional CUDA search backend" OFF)
find_package(OpenMP)
add_library(h3d src/router.cpp src/target_router.cpp)
target_include_directories(h3d PUBLIC include hardware)
target_compile_options(h3d PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
if(OpenMP_CXX_FOUND)
  target_link_libraries(h3d PUBLIC OpenMP::OpenMP_CXX)
endif()
if(H3D_CUDA)
  enable_language(CUDA)
  target_sources(h3d PRIVATE src/search_cuda.cu src/search_cuda_dense.cu src/search_target_cuda.cu)
  target_compile_definitions(h3d PUBLIC H3D_CUDA=1)
  set_target_properties(h3d PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON
    CUDA_SEPARABLE_COMPILATION ON CUDA_RESOLVE_DEVICE_SYMBOLS ON)
endif()
add_executable(h3d_route src/main.cpp)
target_link_libraries(h3d_route PRIVATE h3d)
add_executable(test_router tests/test_router.cpp)
target_link_libraries(test_router PRIVATE h3d)
add_executable(test_cost tests/test_cost.cpp hardware/candidate_cost.cpp)
target_include_directories(test_cost PRIVATE hardware)
enable_testing()
add_test(NAME router COMMAND test_router)
add_test(NAME cost COMMAND test_cost)
add_executable(test_target tests/test_target.cpp)
target_link_libraries(test_target PRIVATE h3d)
add_test(NAME target_oracle COMMAND test_target)
if(H3D_CUDA)
  add_executable(test_cuda tests/test_cuda.cpp)
  target_link_libraries(test_cuda PRIVATE h3d)
  add_test(NAME cuda_incremental COMMAND test_cuda)
endif()
