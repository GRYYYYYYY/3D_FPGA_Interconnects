cmake_minimum_required(VERSION 3.16)
project(lookahead_accel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-ffp-contract=off> $<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math>)
include_directories(include)
find_package(OpenMP REQUIRED)
add_executable(cpu_reference cpu_reference.cpp)
target_link_libraries(cpu_reference PRIVATE OpenMP::OpenMP_CXX)
add_executable(cpu_benchmark cpu_benchmark.cpp)
target_link_libraries(cpu_benchmark PRIVATE OpenMP::OpenMP_CXX)
add_executable(spill_profile tests/spill_profile.cpp)
add_library(lookahead_cpu_backend SHARED cpu_backend.cpp)
target_link_libraries(lookahead_cpu_backend PRIVATE OpenMP::OpenMP_CXX)
add_executable(backend_replay backend_replay.cpp)
target_link_libraries(backend_replay PRIVATE ${CMAKE_DL_LIBS})
add_executable(hls_replay hls_replay.cpp hls/lookahead_top.cpp)
add_executable(semantic_tests tests/testbench.cpp hls/lookahead_top.cpp)
target_compile_options(semantic_tests PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME strict_semantics COMMAND semantic_tests)

foreach(variant heap31 label64)
 add_executable(randomized_${variant} tests/randomized.cpp hls/variants/lookahead_cached.cpp)
 target_compile_options(randomized_${variant} PRIVATE -UNDEBUG)
 if(variant STREQUAL "heap31")
  target_compile_definitions(randomized_${variant} PRIVATE LA_HEAP_TOP=31 LA_LABEL_LINES=0)
 else()
  target_compile_definitions(randomized_${variant} PRIVATE LA_HEAP_TOP=0 LA_LABEL_LINES=64)
 endif()
 add_test(NAME randomized_${variant} COMMAND randomized_${variant})
endforeach()

foreach(policy 0 1)
 foreach(kind randomized testbench)
  set(name packed3_${policy}_${kind})
  add_executable(${name} tests/${kind}.cpp tests/packed_adapter3.cpp hls/variants/lookahead_packed_resident3.cpp)
  target_include_directories(${name} PRIVATE hls/variants)
  target_compile_definitions(${name} PRIVATE LA_HEAP_TOP=127 LA_LABEL_LINES=64 LA_COST_LINES=64 LA_LABEL_WRITEBACK=${policy})
  target_compile_options(${name} PRIVATE -UNDEBUG)
  add_test(NAME ${name} COMMAND ${name})
 endforeach()
endforeach()
