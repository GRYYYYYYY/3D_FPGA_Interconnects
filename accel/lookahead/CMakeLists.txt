cmake_minimum_required(VERSION 3.16)
project(lookahead_accel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-ffp-contract=off> $<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math>)
include_directories(include)
find_package(OpenMP REQUIRED)
add_executable(cpu_reference cpu_reference.cpp)
target_link_libraries(cpu_reference PRIVATE OpenMP::OpenMP_CXX)
add_library(lookahead_cpu_backend SHARED cpu_backend.cpp)
target_link_libraries(lookahead_cpu_backend PRIVATE OpenMP::OpenMP_CXX)
add_executable(backend_replay backend_replay.cpp)
target_link_libraries(backend_replay PRIVATE ${CMAKE_DL_LIBS})
add_executable(hls_replay hls_replay.cpp hls/lookahead_top.cpp)
add_executable(semantic_tests tests/testbench.cpp hls/lookahead_top.cpp)
target_compile_options(semantic_tests PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME strict_semantics COMMAND semantic_tests)

foreach(variant heap31 label64)
 add_executable(randomized_${variant} tests/randomized.cpp hls/variants/lookahead_cached.cpp)
 target_compile_options(randomized_${variant} PRIVATE -UNDEBUG)
 if(variant STREQUAL "heap31")
  target_compile_definitions(randomized_${variant} PRIVATE LA_HEAP_TOP=31 LA_LABEL_LINES=0)
 else()
  target_compile_definitions(randomized_${variant} PRIVATE LA_HEAP_TOP=0 LA_LABEL_LINES=64)
 endif()
 add_test(NAME randomized_${variant} COMMAND randomized_${variant})
endforeach()

foreach(contexts 1 2 4 8)
 add_executable(interleaved_${contexts} tests/interleaved_randomized.cpp hls/variants/lookahead_interleaved_v3.cpp)
 target_compile_definitions(interleaved_${contexts} PRIVATE LA_CONTEXTS=${contexts})
 target_compile_options(interleaved_${contexts} PRIVATE -UNDEBUG)
 add_test(NAME interleaved_${contexts} COMMAND interleaved_${contexts})
endforeach()
add_executable(interleaved_semantics tests/testbench.cpp hls/variants/lookahead_interleaved_v3.cpp)
target_compile_definitions(interleaved_semantics PRIVATE LA_CONTEXTS=1)
target_compile_options(interleaved_semantics PRIVATE -UNDEBUG)
add_test(NAME interleaved_semantics COMMAND interleaved_semantics)

add_executable(resident_randomized tests/randomized.cpp hls/variants/lookahead_resident.cpp)
target_compile_definitions(resident_randomized PRIVATE LA_HEAP_TOP=127 LA_LABEL_LINES=64 LA_COST_LINES=64)
target_compile_options(resident_randomized PRIVATE -UNDEBUG)
add_test(NAME resident_randomized COMMAND resident_randomized)

option(LA_CUDA "Build real CUDA backend; no CPU fallback" OFF)
if(LA_CUDA)
 enable_language(CUDA)
 add_executable(lookahead_cuda cuda/lookahead_cuda.cu)
 set_target_properties(lookahead_cuda PROPERTIES CUDA_STANDARD 14)
 target_compile_options(lookahead_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
 add_library(lookahead_cuda_backend SHARED cuda/cuda_backend.cu)
 set_target_properties(lookahead_cuda_backend PROPERTIES CUDA_STANDARD 14)
 target_compile_options(lookahead_cuda_backend PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
endif()
